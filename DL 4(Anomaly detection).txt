# DL practical 4
import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow.keras import layers, Model
from tensorflow.keras.datasets import mnist

# Load and preprocess data
(x_train, y_train), (x_test, y_test) = mnist.load_data()
x_train = x_train.reshape(-1, 784).astype("float32") / 255
x_test = x_test.reshape(-1, 784).astype("float32") / 255

normal = x_train[y_train == 0]
y_true = (y_test != 0).astype("int32")

# Encoder
i = layers.Input(shape=(784,))
e = layers.Dense(128, activation="relu")(i)
e = layers.Dense(64, activation="relu")(e)
e = layers.Dense(32, activation="relu")(e)

# Decoder
d = layers.Dense(64, activation="relu")(e)
d = layers.Dense(128, activation="relu")(d)
o = layers.Dense(784, activation="sigmoid")(d)

# Autoencoder
model = Model(i, o)
model.compile(optimizer="adam", loss="mse", metrics=["mae"])

# Train
history = model.fit(normal, normal, epochs=10, batch_size=128,
                    validation_split=0.1, shuffle=True, verbose=1)

# Test loss and MAE
loss, mae = model.evaluate(x_test, x_test, verbose=0)
print("Test reconstruction loss:", loss)
print("Test reconstruction MAE:", mae)

# Reconstruction error and threshold
reconstructed = model.predict(x_test, verbose=0)
error = np.mean((x_test - reconstructed)**2, axis=1)

normal_reconstructed = model.predict(normal, verbose=0)
normal_error = np.mean((normal - normal_reconstructed)**2, axis=1)
threshold = np.percentile(normal_error, 95)

# Anomaly detection
y_pred = (error > threshold).astype("int32")

# Performance metrics
tp = np.sum((y_pred == 1) & (y_true == 1))
tn = np.sum((y_pred == 0) & (y_true == 0))
fp = np.sum((y_pred == 1) & (y_true == 0))
fn = np.sum((y_pred == 0) & (y_true == 1))

accuracy = (tp + tn) / len(y_true)
precision = tp / (tp + fp + 1e-8)
recall = tp / (tp + fn + 1e-8)

print("Anomaly threshold:", threshold)
print("Detected anomalies:", np.sum(y_pred))
print("Detection accuracy:", accuracy * 100)
print("Precision:", precision)
print("Recall:", recall)

# Loss plot
plt.plot(history.history["loss"], label="Training Loss")
plt.plot(history.history["val_loss"], label="Validation Loss")
plt.title("Autoencoder Loss")
plt.xlabel("Epoch")
plt.ylabel("MSE")
plt.legend()
plt.grid()
plt.show()
